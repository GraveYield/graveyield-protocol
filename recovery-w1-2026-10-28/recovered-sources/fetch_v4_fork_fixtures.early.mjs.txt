    48→// ---------------------------------------------------------------- base58
    49→const B58 = "123456789ABCDEFGHJKLMNPQRSTUVWXYZabcdefghijkmnopqrstuvwxyz";
    50→function b58encode(bytes) {
    51→  if (bytes.length === 0) return "";
    52→  let digits = [0];
    53→  for (const byte of bytes) {
    54→    let carry = byte;
    55→    for (let j = 0; j < digits.length; j++) {
    56→      carry += digits[j] << 8;
    57→      digits[j] = carry % 58;
    58→      carry = (carry / 58) | 0;
    59→    }
    60→    while (carry > 0) {
    61→      digits.push(carry % 58);
    62→      carry = (carry / 58) | 0;
    63→    }
    64→  }
    65→  for (let i = 0; bytes[i] === 0 && i < bytes.length - 1; i++) digits.push(0);
    66→  return digits
    67→    .reverse()
    68→    .map((d) => B58[d])
    69→    .join("");
    70→}

    72→// ---------------------------------------------------------------- rpc
    73→let callCount = 0;
    74→async function rpc(method, params) {
    75→  callCount++;
    76→  if (callCount > 1) await new Promise((r) => setTimeout(r, 350));
    77→  for (let attempt = 1; attempt <= 4; attempt++) {
    78→    let res;
    79→    try {
    80→      res = await fetch(RPC, {
    81→        method: "POST",
    82→        headers: { "Content-Type": "application/json" },
    83→        body: JSON.stringify({ jsonrpc: "2.0", id: 1, method, params }),
    84→      });
    85→    } catch (e) {
    86→      if (attempt === 4) throw new Error(`RPC fetch failed: ${e.message}`);
    87→      await new Promise((r) => setTimeout(r, 1500 * attempt));
    88→      continue;
    89→    }
    90→    if (res.status === 429 || res.status >= 500) {
    91→      if (attempt === 4) throw new Error(`RPC ${method} HTTP ${res.status} after retries`);
    92→      await new Promise((r) => setTimeout(r, 2000 * attempt));
    93→      continue;
    94→    }
    95→    const json = await res.json();
    96→    if (json.error) throw new Error(`RPC ${method} error: ${JSON.stringify(json.error)}`);
    97→    return json.result;
    98→  }
    99→}

   101→async function getAccount(addr) {
   102→  const r = await rpc("getAccountInfo", [addr, { encoding: "base64" }]);
   103→  if (!r || !r.value) return null;
   104→  return {
   105→    pubkey: addr,
   106→    lamports: r.value.lamports,
   107→    owner: r.value.owner,
   108→    data: Buffer.from(r.value.data[0], "base64"),
   109→  };
   110→}

   112→// ---------------------------------------------------------------- readers
   113→function u64le(buf, off) {
   114→  return BigInt.asUintN(64, buf.readBigUInt64LE(off));
   115→}
   116→function pk(buf, off) {
   117→  return b58encode(buf.subarray(off, off + 32));
   118→}

   120→// ---------------------------------------------------------------- program ELF
   121→async function getProgramElf(programId, label) {
   122→  const prog = await getAccount(programId);
   123→  if (!prog) throw new Error(`${label} program ${programId} not found`);
   124→  if (prog.data.length > 1024 && prog.data[0] === 0x7f) {
   125→    console.log(`  ${label}: direct ELF ${prog.data.length} bytes`);
   126→    return prog.data;
   127→  }
   128→  if (prog.data.length < 36)
   129→    throw new Error(`${label}: program account too small (${prog.data.length})`);
   130→  // Upgradeable loader Program layout: [u32 variant][programdata pubkey] = 36 bytes.
   131→  const dataAddr = pk(prog.data, 4);
   132→  const dataAcct = await getAccount(dataAddr);
   133→  if (!dataAcct) throw new Error(`${label}: programdata ${dataAddr} not found`);
   134→  let off = -1;
   135→  for (let i = 0; i < 256; i++) {
   136→    if (
   137→      dataAcct.data[i] === 0x7f &&
   138→      dataAcct.data[i + 1] === 0x45 &&
   139→      dataAcct.data[i + 2] === 0x4c &&
   140→      dataAcct.data[i + 3] === 0x46
   141→    ) {
   142→      off = i;
   143→      break;
   144→    }
   145→  }
   146→  if (off < 0) throw new Error(`${label}: no ELF magic in programdata`);
   147→  console.log(
   148→    `  ${label}: ELF @ programdata+${off} (${dataAcct.data.length - off} bytes, programdata ${dataAddr})`,
   149→  );
   150→  return dataAcct.data.subarray(off);
   151→}
