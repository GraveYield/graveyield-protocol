   199→async function expectRevert({
   200→  id,
   201→  name,
   202→  ixs,
   203→  signers,
   204→  expectCode,
   205→  expectFamily,
   206→  feePayerBudgetLamports = MAX_FAILED_TX_FEE_LAMPORTS,
   207→  pdaMustStayAbsent = [],
   208→  tokenAccounts = {},
   209→}) {
   210→  const feePayer = signers[0].publicKey;
   211→  const beforeFee = await solBalance(connection, feePayer);
   212→  const tokensBefore = await snapshotTokenAccounts(connection, tokenAccounts);
   213→  const res = await sendTx(connection, ixs, signers);
   214→  const afterFee = await solBalance(connection, feePayer);
   215→  const tokensAfter = await snapshotTokenAccounts(connection, tokenAccounts);
   216→  const code = customCodeFromErr(res.err);
   217→  const family = code === null && res.err ? errFamilyFromErr(res.err, res.logs) : null;
   218→  const pass =
   219→    expectCode !== undefined ? code === expectCode : family === expectFamily && res.err != null;
   220→  const feeDelta = beforeFee - afterFee;
   221→  const feeOk = feeDelta >= 0n && feeDelta <= feePayerBudgetLamports;
   222→  const tokensSame = snapshotDiffZero(tokensBefore, tokensAfter);
   223→  const absentChecks = [];
   224→  for (const [label, pda] of pdaMustStayAbsent) {
   225→    absentChecks.push({ label, absent: (await connection.getAccountInfo(pda)) === null });
   226→  }
   227→  const zeroMovement = {
   228→    proved: feeOk && tokensSame && absentChecks.every((c) => c.absent),
   229→    fee_payer_delta_lamports: feeDelta.toString(),
   230→    fee_ok: feeOk,
   231→    tokens_unchanged: tokensSame,
   232→    pdas_absent: absentChecks,
   233→    note:
   234→      "fee-only debit on the submitter is the expected real cost of a failed transaction and is recorded, not normalized away; every other tracked balance and the settlement PDA set must be unchanged",
   235→  };
   236→  report.adversarial({
   237→    id,
   238→    name,
   239→    expected: expectCode !== undefined ? { code: expectCode } : { family: expectFamily },
   240→    observed: { code, family, err: res.err ?? null },
   241→    pass,
   242→    tx: res.sig,
   243→    slot: res.slot,
   244→    logs_tail: (res.logs ?? []).slice(-6),
   245→    zero_movement: zeroMovement,
   246→  });
   247→  if (!pass) {
   248→    failFast(
   249→      `adversarial ${id} (${name}) did not revert as expected: err=${JSON.stringify(res.err)} logs=${(res.logs ?? []).slice(-4)}`,
   250→    );
   251→  }
   252→  if (!zeroMovement.proved) {
   253→    failFast(`adversarial ${id} (${name}) reverted but the zero-state-movement invariant failed`);
   254→  }
   255→  return { code, family, res };
   256→}

   258→// C1 attestation: real local slot anchor + fork-precedented synthetic
   259→// last-swap timestamp (documented boundary). issuedSlot points a few slots
   260→// back so its block (and slot hash) is always resolvable.
   261→async function buildC1Attestation({ oracleKeypair, lastSwapUnixTs }) {
   262→  const tip = Number(await connection.getSlot());
   263→  const issuedSlot = Math.max(1, tip - 4);
   264→  const slotHash = await SDK.fetchSlotHash(connection, issuedSlot);
   265→  if (!slotHash) throw new Error(`no slot hash for slot ${issuedSlot}`);
   266→  const msg = SDK.buildAttestationMessage({
   267→    ammProgramId: IDS.raydiumV4,
   268→    poolAddress: POOL,
   269→    lastSwapUnixTs: BigInt(lastSwapUnixTs),
   270→    issuedSlot: BigInt(issuedSlot),
   271→    slotHash,
   272→  });
   273→  if (msg.length !== 112) throw new Error(`C1 msg must be 112B, got ${msg.length}`);
   274→  const signature = nacl.sign.detached(msg, oracleKeypair.secretKey);
   275→  return { msg, signature, issuedSlot, slotHash };
   276→}

   348→// Non-canonical ed25519 precompile: runtime-valid (signature verifies) but
   349→// sig offset 56 instead of the canonical 48 — the scanner's offset pin
   350→// (ORACLE_PIPELINE.md §2.3 check order) reverts 6025.
   351→function shiftedPrecompileIx({ signature, oraclePublicKey, message, scannerInstructionIndex }) {
   352→  const data = Buffer.alloc(120); // header 16 + pk 32 (@16) + sig 64 (@56)
   353→  data[0] = 1;
   354→  const w = (off, v) => data.writeUInt16LE(v, off);
   355→  w(2, 56); // signature_offset — SHIFTED (canonical pin is 48)
   356→  w(4, 0xffff); // signature_instruction_index
   357→  w(6, 16); // public_key_offset
   358→  w(8, 0xffff); // public_key_instruction_index
   359→  w(10, 72); // message_data_offset (scanner ix data @72)
   360→  w(12, message.length);
   361→  w(14, scannerInstructionIndex);
   362→  Buffer.from(oraclePublicKey.toBytes()).copy(data, 16);
   363→  Buffer.from(signature).copy(data, 56);
   364→  return new TransactionInstruction({
   365→    programId: SDK.ED25519_PROGRAM_ID ?? new PublicKey("Ed25519SigVerify111111111111111111111111111"),
   366→    keys: [],
   367→    data,
   368→  });
   369→}
