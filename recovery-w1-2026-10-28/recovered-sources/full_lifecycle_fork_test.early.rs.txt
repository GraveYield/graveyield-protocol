     3→// Phase 6 — the first COMPLETE GraveYield integration test (roadmap
     4→// Phase 6). One test executes the entire protocol lifecycle end to end
     5→// against real mainnet Raydium V4 / OpenBook / SPL-token bytecode:
     6→//
     7→//   candidate pool (real mainnet AmmInfo state)
     8→//     -> GraveScanner initialize (governance thresholds, oracle keys)
     9→//     -> record_launch_price      (C2 baseline: oracle-signed 168B Ed25519
    10→//                                  attestation, runtime precompile path)
    11→//     -> evaluate_pool_phase_1    (C1 evidence: indexer-signed 112B Ed25519
    12→//                                  attestation + six criteria over the real
    13→//                                  pool bytes) -> EligibilityAnchor
    14→//     -> multi-epoch waiting period (VM warp; >= MIN_EPOCH_CONFIRMATION)
    15→//     -> evaluate_pool_phase_2    (fresh C1 attestation, bitmap equality,
    16→//                                  epoch gap) -> EligibilityCert
    17→//     -> LP-holder snapshot over the live VM ledger (grave-snapshotter)
    18→//     -> SnapshotMerkleTree -> sealed SnapshotArtifact -> JSON publication
    19→//     -> salvage_pool             (real withdraw CPI + Jupiter stand-in
    20→//                                  conversion + 40/40/20 settlement; the
    21→//                                  artifact root + supply sealed on chain)
    22→//     -> claim_lp_proceeds x5     (artifact proofs -> SOL in wallets)
    23→//
    24→// and asserts every state transition along the way.
    25→//
    26→// The forged-EligibilityCert stand-in the Phase 2.1-5.3 suites documented
    27→// is GONE here: the cert that authorizes salvage is issued by the real
    28→// grave-scanner program running as BPF in the same VM, and both
    29→// attestation paths execute through the runtime's actual ed25519
    30→// precompile verification. (Phase 6 also CORRECTED the scanner's
    31→// precompile offset contract — the previously shipped 14-byte header is
    32→// not a wire format any Solana runtime accepts; see
    33→// grave-scanner/src/attestation.rs and spec rev 1.10.0.)
    34→//
    35→// Harness notes:
    36→//   * The withdraw-leg economics are measured on a disposable VM that
    37→//     runs the SAME full scanner path (no forged cert anywhere) with the
    38→//     conversion leg disabled — the VM is deterministic, so the real
    39→//     salvage's withdraw produces byte-identical deltas (the Phase 5.3
    40→//     suite's measurement pattern, upgraded to the honest boot).
    41→//   * Governance thresholds are configured explicitly (they are
    42→//     governance parameters, not constants): the fork pool is a real
    43→//     mainnet account, and the C1/C2 evidence authorities are test
    44→//     keypairs, exactly as in production the indexer/oracle keys are
    45→//     registered in ProtocolConfig at initialize.

   165→/// Real mainnet account from a manifest section: (pubkey, Account).
   166→fn load_real_account(section: &serde_json::Value, name: &str) -> (Pubkey, Account) {
   167→    let entry = &section[name];
   168→    let data = std::fs::read(fixtures_dir().join(entry["file"].as_str().unwrap()))
   169→        .unwrap_or_else(|e| panic!("missing fixture for {name}: {e}"));
   170→    let account = Account {
   171→        lamports: entry["lamports"].as_u64().unwrap(),
   172→        data,
   173→        owner: pk(entry["owner"].as_str().unwrap()),
   174→        executable: false,
   175→        rent_epoch: u64::MAX,
   176→    };
   177→    (pk(entry["pubkey"].as_str().unwrap()), account)
   178→}

   226→fn forged_token_account(mint: &Pubkey, owner: &Pubkey, amount: u64) -> Account {
   227→    let mut data = vec![0u8; 165];
   228→    data[0..32].copy_from_slice(mint.as_ref());
   229→    data[32..64].copy_from_slice(owner.as_ref());
   230→    data[TA_AMOUNT..TA_AMOUNT + 8].copy_from_slice(&amount.to_le_bytes());
   231→    data[108] = 1; // AccountState::Initialized
   232→    Account {
   233→        lamports: Rent::default().minimum_balance(165),
   234→        data,
   235→        owner: spl_token_id(),
   236→        executable: false,
   237→        rent_epoch: u64::MAX,
   238→    }

   478→fn wallet_ata(wallet: &Pubkey, mint: &Pubkey) -> Pubkey {
   479→    Pubkey::find_program_address(

   486→/// The four non-salvor claimants: deterministic splits of the claimable
   487→/// remainder after the pool-LP custody share and the burn plan. The salvor
   488→/// is claimant #0 (D11 policy 2: the pre-burn balance is an ordinary leaf).
   489→fn claim_wallets(claimable_total: u64) -> Vec<Claimant> {
   490→    let b1 = claimable_total / 2;
   491→    let b2 = claimable_total * 3 / 10;
   492→    let b3 = claimable_total / 8;
   493→    let b4 = claimable_total - b1 - b2 - b3;
   494→    [
   495→        (b1, LAMPORTS_PER_SOL),
   496→        (b2, LAMPORTS_PER_SOL),
   497→        (b3, LAMPORTS_PER_SOL),
   498→        (b4, LAMPORTS_PER_SOL),
   499→    ]
   500→    .into_iter()
   501→    .map(|(balance, start_lamports)| Claimant {
   502→        wallet: Keypair::new(),
   503→        balance,
   504→        start_lamports,
   505→    })
   506→    .collect()
   507→}
   508→
   509→/// Genesis accounts for the claim side: the pool-LP custody token account
   510→/// (owner = the real AMM authority — can never sign a claim, the exact
   511→/// shape the D11 sink exclusion exists for), each wallet's LP ATA (so the
   512→/// VM ledger is enumerable and Σ == supply holds) and each wallet's
   513→/// funding (fees + ClaimRecord rent).
   514→fn claim_side_genesis(
   515→    lp_mint: &Pubkey,
   516→    custody_account: Pubkey,
   517→    custody_balance: u64,
   518→    wallets: &[Claimant],
   519→) -> Vec<(Pubkey, Account)> {
   520→    let mut out = vec![(
   521→        custody_account,
   522→        forged_token_account(lp_mint, &RAYDIUM_V4_AMM_AUTHORITY, custody_balance),
   523→    )];
   524→    for claimant in wallets {
   525→        out.push((
   526→            wallet_ata(&claimant.wallet.pubkey(), lp_mint),
   527→            forged_token_account(lp_mint, &claimant.wallet.pubkey(), claimant.balance),
   528→        ));
   529→        out.push((
   530→            claimant.wallet.pubkey(),
   531→            Account {
   532→                lamports: claimant.start_lamports,
   533→                data: vec![],
   534→                owner: system_program::ID,
   535→                executable: false,
   536→                rent_epoch: u64::MAX,
   537→            },
   538→        ));
   539→    }

   657→/// (slot, epoch, unix_timestamp) from the live Clock sysvar.
   658→async fn read_clock(client: &mut BanksClient) -> (u64, u64, i64) {
   659→    let a = acct(client, &clock_sysvar::id()).await;
   660→    let slot = read_u64(&a.data, 0);
   661→    let epoch = read_u64(&a.data, 16);
   662→    let ts = i64::from_le_bytes(a.data[32..40].try_into().unwrap());
   663→    (slot, epoch, ts)
   664→}

   666→/// The freshest (slot, hash) entry the VM's SlotHashes sysvar can resolve.
   667→/// Entries are scanned (newest-first in practice) and the max slot is
   668→/// returned; fails the test when no positive-slot entry exists (slot 0 is
   669→/// unusable — the scanner requires `issued_slot > 0`).
   670→async fn newest_slot_hash(client: &mut BanksClient, now_slot: u64) -> (u64, [u8; 32]) {
   671→    let a = acct(client, &slot_hashes_sysvar::id()).await;
   672→    let count = read_u64(&a.data, 0) as usize;
   673→    let mut best: Option<(u64, [u8; 32])> = None;
   674→    for i in 0..count {
   675→        let base = 8 + i * 40;
   676→        let slot = read_u64(&a.data, base);
   677→        if slot == 0 || slot > now_slot {
   678→            continue;
   679→        }
   680→        let mut h = [0u8; 32];
   681→        h.copy_from_slice(&a.data[base + 8..base + 40]);
   682→        if best.is_none_or(|(s, _)| slot > s) {
   683→            best = Some((slot, h));
   684→        }
   685→    }
   686→    best.unwrap_or_else(|| panic!("no usable slot-hash entry at slot {now_slot}"))
   687→}
